    \begin{itemize}[leftmargin=1em, itemsep=1pt]

\textbf{$\rightarrow$ \underline{Basic Isabelle Interface}:}
\begin{itemize}[leftmargin=1.2em, itemsep=3pt]
    \item \textbf{Manipulating theory file} (\texttt{.thy})
    \item \textbf{Writing terms:}
    \begin{itemize}[leftmargin=1em, itemsep=1pt]
        \item Numbers ($\text{nat}, \text{int}, \text{real}$)
        \item Lists (\texttt{[]}, \texttt{\#}, \texttt{@})
        \item Sets (\texttt{\{x. P x\}}, \texttt{$\cup$}, \texttt{$\cap$})
        \item Products / tuples (\texttt{(a, b)})
        \item Functions ($f :: \alpha \to \beta$)
    \end{itemize}
    \item \textbf{Type annotations} for all these terms (\texttt{t :: type})
\end{itemize}
\end{multicols}
\end{tcolorbox}
